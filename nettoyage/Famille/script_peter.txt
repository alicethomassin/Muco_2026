
----- start of step 1 ----------------------------------------------------------
C4_P0 <- C4_B5 %>% 
  
# sf_interromp is the variable for the question F-8 "Has this relationship ended?".
# I changed the values from character to integer to help arrange the relationship's orders.

  mutate(
  sf_interromp = case_when(
    sf_interromp == "0" ~ 0L,
    sf_interromp == "1" ~ 1L,
    TRUE ~ as.integer(sf_interromp)
    )
  ) %>%
  
# sf_conjt_an_union is the variable for the question F-6 "When did this relationship start?".
# I ordered each relationship in chronological order.

  arrange(id_anonymat, sf_conjt_an_union, desc(sf_interromp)) %>%

# I grouped the data by the id_anonymat and I created the variable rang_cpl.
# to show the rank of every relationship based on their position in the grouped data.
  
  mutate(
    .by = id_anonymat,
    rang_cpl = row_number(),
    .after = id_anonymat
    ) %>% 
    
# I changed the rank from 1 to 0 for those who have not had a relationship.

  mutate(
    rang_cpl = replace_when(
      rang_cpl,
      nb_cpl == 0 ~ 0
    )
  ) %>% 
  
# I filtered only those who had a rank of 1 or 0.

  filter(rang_cpl <= 1)

_____ end of step 1 __________


----- start of step 2 ----------------------------------------------------------

C4_P1 <- C4_P0 %>% 

  mutate(
  
# Create binary variable to differentiate those who have had a relationship
# from those who haven't.
  
    relationship = case_when(
      rang_cpl == 0 ~ "no",
      rang_cpl == 1 ~ "yes",
      TRUE ~ "E"
    ),
    
# Create variable for the nature of the relationship based on the question F-5
# "Nature of the relationship". I added the value "NR" for "No Relationship".
    
    nature = case_when(
      sf_nature_union == "1" ~ "married",
      sf_nature_union == "2" ~ "cohabiting",
      sf_nature_union == "3" ~ "living appart",
      relationship == "no" ~ "NR",
      relationship == "yes" & is.na(sf_nature_union) ~ "skipped F-5",
      TRUE ~ "E"
    ),
    
# The date for the start of the relationship has been obtained by combining
# sf_conjt_an_union (the year) and sf_conjt_mois_union (the month). 
# If a CPF did not give the month or the year, either or both value will be 
# NA (2000-NA, NA-11, NA-NA). 
# If a CFP did not have a relationship, their value is "NR".
    
    date_start = case_when(
      relationship == "no" ~ "NR",
      relationship == "yes" ~ sprintf("%d-%02d", sf_conjt_an_union, sf_conjt_mois_union)),
      
# The same approach was used for the separation date. 
# If the relationship has not ended, the value is "ongoing". 
# If the relationship has ended (sf_interromp == 1) but the CFP has not given the
# month or the year it ended, the value will contain NAs (2000-NA, NA-11, NA-NA).
# If the CFP did not answer the question F-8 "Has this relationship ended?", 
# the value is "skipped F-8".
      
    date_end = case_when(
      relationship == "no" ~ "NR",
      sf_interromp == 0 ~ "ongoing",
      sf_interromp == 1 ~ sprintf("%d-%02d", sf_an_interromp, sf_mois_interromp),
      relationship == "yes" & is.na(sf_interromp) ~ "skipped F-8"
    ),
    
# Same approach for the missing values (NR or skipped F-3 "Sex of the partner").
    
    sex_partner = case_when(
      relationship == "no" ~ "NR",
      sf_conjt_sexe == "1" ~ "male",
      sf_conjt_sexe == "2" ~ "female",
      relationship == "yes" & is.na(sf_conjt_sexe) ~ "skipped F-3",
      TRUE ~ "E"
    ),
    .after = id_anonymat
  )

_____ end of step 2 __________


----- start of step 3 ----------------------------------------------------------

C4_P2 <- C4_P1 %>% 

# Select the 6 columns of interest.

  select(id_anonymat, relationship, sex_partner, nature, date_start, date_end)
